# BAU Dashboard, Abandon API and BD-1 Line Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `bau.home.nakomis.com` shows the PRs awaiting Martin, the deploy list and the last run from `state.json`, offers one action (abandon), and BD-1 shows a BAU line in the menu bar.

**Architecture:** A FastAPI service in the `bau` package serves the static dashboard and `POST /abandon/{ref}`; it reads `state.json` for everything else. The dashboard is a Vite + React + Tailwind 4 app with no state of its own. The abandon action is a plain function in `bau/abandon.py` that the API and a `bau abandon REF` command both call. BD-1's line is one more `(Health, String)` producer in the Rust tray, fed by a poll of `https://bau.home.nakomis.com/state.json` over the existing mTLS client.

**Tech Stack:** Python 3.12, FastAPI, uvicorn, pytest; Vite 8, React 19, Tailwind 4, Biome, Vitest, pnpm 11; Rust (BD-1 in `home-infra/mac/conversation-memory-forwarder`).

**Spec:** `docs/superpowers/specs/2026-10-14-bau-design.md`, sections 8 and 2 (BD-1 thresholds). Depends on Plan 1 (`2026-10-14-bau-core.md`) being merged.

## Global Constraints

- Same repo, licence, British English and commit rules as Plan 1. Dashboard work is in `web/` of `nakomis/bau`; BD-1 work is a PR against `nakomis/home-infra`.
- Web toolchain copied from `~/repos/nakomis/nakostat/web`: `packageManager: pnpm@11.5.1`, Vite with `@tailwindcss/vite`, Biome config verbatim, Vitest in `vite.config.ts` with a 70% line threshold, `pnpm-workspace.yaml` with `minimumReleaseAge: 1440`. No shadcn, oRPC, router or query libraries: one page, three sections, plain `fetch`.
- The API binds `127.0.0.1:8800` on Cere only; Leia's nginx does mTLS in front (Plan 3). The API itself has no auth.
- The snapshot schema is Plan 1's `build_snapshot` output (`schema: 1`). The dashboard must render an empty or missing snapshot without crashing.
- BD-1 colours: green `n < 10`, amber `10 <= n < 30`, red `n >= 30`; red on `last_run.stop_reason` in `("rejected", "failed")` or `last_run` null after the first run; red on `generated_at` older than 26 hours; amber on `telemetry == "missing"`.
- Runtime deps added to `pyproject.toml`: `fastapi>=0.115`, `uvicorn>=0.30`. Dev: `httpx` is already present and is FastAPI's test client transport.

- [ ] **Step 3: Implement `bau/abandon.py`**

```python
from __future__ import annotations

from collections.abc import Callable
from dataclasses import dataclass

from bau.checkout import Repos
from bau.config import ProjectConfig
from bau.github import GitHubClient
from bau.plane import PlaneClient, split_ref
from bau.runloop import Night, previous_state


class AlreadyAbandoned(RuntimeError):
    pass


@dataclass(frozen=True)
class AbandonResult:
    ref: str
    pr: int | None
    restored_state: str


def abandon(
    ref: str,
    *,
    plane: PlaneClient,
    github_factory: Callable[[str], GitHubClient],
    repos: Repos,
    projects: dict[str, ProjectConfig],
    worker_login: str,
    note: str = "",
) -> AbandonResult:
    comments = plane.comments(ref)
    last_pick = max((c.created_at for c in comments if "[bau:picked-up" in c.text), default="")
    last_abandon = max((c.created_at for c in comments if "[bau:abandoned]" in c.text), default="")
    if not last_pick or last_abandon > last_pick:
        raise AlreadyAbandoned(f"{ref} has no active BAU pick-up to abandon")
    restored = previous_state(comments) or "New"

    item = plane.get_item(ref)
    ident, _ = split_ref(ref)
    project = projects.get(ident)
    pr_number = None
    branch = None
    if project:
        gh = github_factory(project.repo)
        for pr in gh.open_prs(project.repo, worker_login):
            if Night.ref_of(pr) == ref:
                gh.comment(
                    project.repo,
                    pr.number,
                    "[bau] Abandoned by Martin from the BAU dashboard." + (f" Note: {note}" if note else ""),
                )
                gh.close_pr(project.repo, pr.number)
                pr_number, branch = pr.number, pr.head_ref
                break
    if branch is None:
        for c in comments:
            if "[bau:picked-up" in c.text and "`" in c.text:
                branch = c.text.split("`")[1]
    if branch:
        repos.delete_checkout(branch)
    plane.set_state(ref, restored)
    plane.remove_labels(ref, ["BAU"])
    plane.comment(
        ref,
        f"[bau:abandoned] by Martin. Returned to {restored}; BAU label removed."
        + (f" Note: {note}" if note else ""),
    )
    del item
    return AbandonResult(ref, pr_number, restored)
```

Drop the `item = plane.get_item(ref)` and `del item` lines; they were a reminder that the item must exist, which `plane.comments` already guarantees by raising.

```python
from __future__ import annotations

from collections.abc import Callable
from pathlib import Path

from fastapi import FastAPI, HTTPException
from fastapi.responses import FileResponse, JSONResponse
from fastapi.staticfiles import StaticFiles
from pydantic import BaseModel

from bau.abandon import AlreadyAbandoned, abandon
from bau.checkout import Repos
from bau.config import Config, ProjectConfig
from bau.github import GitHubClient
from bau.plane import PlaneClient
from bau.snapshot import read_snapshot


class AbandonBody(BaseModel):
    note: str = ""


def create_app(
    cfg: Config,
    projects: dict[str, ProjectConfig],
    plane: PlaneClient,
    github_factory: Callable[[str], GitHubClient],
    repos: Repos,
    static_dir: Path | None,
) -> FastAPI:
    app = FastAPI(title="bau")

    @app.get("/state.json")
    def state():
        snap = read_snapshot(cfg.paths.snapshot)
        return JSONResponse(
            snap if snap is not None else {"schema": 1, "missing": True},
            headers={"Cache-Control": "no-store"},
        )

    @app.post("/abandon/{ref}")
    def abandon_route(ref: str, body: AbandonBody | None = None):
        try:
            r = abandon(
                ref,
                plane=plane,
                github_factory=github_factory,
                repos=repos,
                projects=projects,
                worker_login=cfg.identities.worker_github_login,
                note=(body.note if body else ""),
            )
        except AlreadyAbandoned as e:
            raise HTTPException(409, str(e)) from e
        return {"ref": r.ref, "pr": r.pr, "restored_state": r.restored_state}

    if static_dir and static_dir.exists():

        @app.get("/")
        def index():
            return FileResponse(static_dir / "index.html")

        app.mount("/", StaticFiles(directory=static_dir), name="static")
    return app


def serve(cfg: Config, projects, plane, github_factory, repos, static_dir: Path, port: int = 8800) -> None:
    import uvicorn

    uvicorn.run(
        create_app(cfg, projects, plane, github_factory, repos, static_dir), host="127.0.0.1", port=port
    )
```

```ts
export function ageLabel(days: number): string {
  if (days <= 0) return 'today';
  return days === 1 ? '1 day' : `${days} days`;
}

export function pct(fraction: number): string {
  return `${Math.round(fraction * 100)}%`;
}

export async function fetchState(): Promise<import('../types').Snapshot> {
  const r = await fetch('/state.json', { cache: 'no-store' });
  return (await r.json()) as import('../types').Snapshot;
}

export async function abandon(ref: string, note: string): Promise<void> {
  const r = await fetch(`/abandon/${encodeURIComponent(ref)}`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ note }),
  });
  if (!r.ok) throw new Error(`abandon failed: ${r.status}`);
}
```

```rust
#[derive(Clone, Debug, PartialEq)]
pub struct BauStatus {
    pub reachable: bool,
    pub open_prs: Option<u32>,
    pub last_run_ok: Option<bool>,
    pub stop_reason: Option<String>,
    pub telemetry_missing: bool,
    pub stale: bool,
    pub error: Option<String>,
}

pub fn bau_status_from(v: &serde_json::Value, now: SystemTime) -> BauStatus {
    let open = v["open_prs"].as_array().map(|a| a.len() as u32);
    let stop = v["last_run"]["stop_reason"].as_str().map(str::to_string);
    let last_run_ok = v["last_run"].is_object().then(|| !matches!(stop.as_deref(), Some("rejected") | Some("failed")));
    let telemetry_missing = v["last_run"]["telemetry"].as_str() == Some("missing");
    let stale = v["generated_at"].as_str()
        .and_then(|s| chrono::DateTime::parse_from_rfc3339(s).ok())
        .map(|t| now.duration_since(SystemTime::from(t)).map(|d| d.as_secs() > 26 * 3600).unwrap_or(false))
        .unwrap_or(true);
    BauStatus { reachable: true, open_prs: open, last_run_ok, stop_reason: stop, telemetry_missing, stale, error: None }
}

pub async fn poll_bau(http: &reqwest::Client, url: &str, now: SystemTime) -> BauStatus {
    let unreachable = |e: String| BauStatus { reachable: false, open_prs: None, last_run_ok: None,
        stop_reason: None, telemetry_missing: false, stale: false, error: Some(e) };
    match http.get(url).send().await {
        Ok(r) if r.status().is_success() => match r.json::<serde_json::Value>().await {
            Ok(v) => bau_status_from(&v, now),
            Err(e) => unreachable(format!("bad json: {e}")),
        },
        Ok(r) => unreachable(format!("http {}", r.status())),
        Err(e) => unreachable(e.to_string()),
    }
}
```

```rust
fn bau_line(b: &BauStatus) -> (Health, String) {
    if !b.reachable {
        let why: String = b.error.as_deref().unwrap_or("unreachable").chars().take(32).collect();
        return (Health::Down, format!("BAU: unreachable ({why})"));
    }
    let n = b.open_prs.unwrap_or(0);
    let run = match (b.last_run_ok, b.stop_reason.as_deref()) {
        (Some(true), Some(r)) => format!("last run ok ({r})"),
        (Some(true), None) => "last run ok".to_string(),
        (Some(false), Some(r)) => format!("last run FAILED ({r})"),
        (Some(false), None) => "last run FAILED".to_string(),
        (None, _) => "no run yet".to_string(),
    };
    let text = format!("BAU: {n} PRs awaiting review · {run}{}", if b.stale { " · stale" } else { "" });
    let health = if b.stale || b.last_run_ok == Some(false) || n >= 30 { Health::Down }
        else if n >= 10 || b.telemetry_missing { Health::Warn }
        else { Health::Ok };
    (health, text)
}
```
